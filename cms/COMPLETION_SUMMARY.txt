================================================================================
                          DATABASE SCHEMA
================================================================================

TABLE: users_profile
  - id (UUID, Primary Key)
  - email (TEXT, Unique)
  - full_name (TEXT)
  - role (TEXT: 'admin' or 'editor')
  - created_at, updated_at

TABLE: careers
  - id, title, description, location, job_type, salary_range
  - requirements, application_url, created_by
  - created_at, updated_at

TABLE: blog_posts
  - id, title, slug, content, excerpt, featured_image_url
  - author_id, status, published_at
  - created_at, updated_at

TABLE: blog_categories
  - id, name, slug, created_at

TABLE: blog_posts_categories
  - post_id (FK), category_id (FK)

TABLE: shop_products
  - id, name, description, price, currency, image_url
  - category, sku, created_by
  - created_at, updated_at

TABLE: gallery_images
  - id, title, description, image_url, image_path
  - gallery_category, sort_order, created_by
  - created_at, updated_at
